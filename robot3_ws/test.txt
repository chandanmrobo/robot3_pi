#!/usr/bin/env python3
import rclpy
from rclpy.node import Node

import cv2
import numpy as np
from sensor_msgs.msg import CompressedImage


class LaptopCamPublisher(Node):
    def __init__(self):
        super().__init__("laptop_cam_test_pub")

        # Publish to the topic camera_node.py expects
        self.pub = self.create_publisher(
            CompressedImage,
            "/robot2/camera/image_raw/compressed",
            10
        )

        # Open laptop webcam (0)
        self.cap = cv2.VideoCapture(0)
        if not self.cap.isOpened():
            self.get_logger().error("Cannot open laptop webcam!")
            raise RuntimeError("Camera not found")

        self.get_logger().info("Laptop webcam streaming â†’ /robot2/camera/...")

        # Publish @ 20 FPS
        self.timer = self.create_timer(1/20, self.publish_frame)

    def publish_frame(self):
        ret, frame = self.cap.read()
        if not ret:
            self.get_logger().warn("Frame capture failed")
            return

        # Resize to match your WebRTC pipeline (optional)
        frame = cv2.resize(frame, (640, 480))

        # Encode JPEG
        ok, jpeg = cv2.imencode(".jpg", frame)
        if not ok:
            return

        msg = CompressedImage()
        msg.format = "jpeg"
        msg.data = jpeg.tobytes()

        self.pub.publish(msg)

    def destroy_node(self):
        self.cap.release()
        super().destroy_node()


def main(args=None):
    rclpy.init(args=args)
    node = LaptopCamPublisher()

    try:
        rclpy.spin(node)
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == "__main__":
    main()
